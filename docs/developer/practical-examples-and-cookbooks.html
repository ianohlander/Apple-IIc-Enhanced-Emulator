1. Full-Stack C# Retro Breakout Arcade Engine

A complete, object-oriented Breakout arcade game in C# utilizing Double Hi-Res graphics, 2D brick collision physics, paddle tracking via Apple II game paddles ($C064), and Mockingboard sound effects: 

using System;
using System.Collections.Generic;
using Apple2.Ultra;

namespace RetroBreakout
{
    public class Brick
    {
        public int X { get; set; }
        public int Y { get; set; }
        public int Color { get; set; }
        public bool IsAlive { get; set; }

        public Brick(int x, int y, int color)
        {
            X = x;
            Y = y;
            Color = color;
            IsAlive = true;
        }

        public void Draw()
        {
            if (!IsAlive) return;
            // Draw a 32x8 pixel Double Hi-Res Brick
            for (int r = 0; r < 8; r++)
            {
                AppleVideo.DrawLine(X, Y + r, X + 30, Y + r, Color);
            }
        }
    }

    public class BreakoutGame
    {
        private List<Brick> _bricks;
        private int _score;
        private int _lives;
        private int _ballX, _ballY;
        private int _ballDx, _ballDy;

        public BreakoutGame()
        {
            _score = 0;
            _lives = 3;
            _ballX = 280;
            _ballY = 120;
            _ballDx = 2;
            _ballDy = -2;
            _bricks = new List<Brick>();

            // Populate 4 rows of 14 bricks (56 bricks total)
            int[] rowColors = { AppleVideo.DeepRed, AppleVideo.Orange, AppleVideo.LightGreen, AppleVideo.Yellow };
            for (int row = 0; row < 4; row++)
            {
                for (int col = 0; col < 14; col++)
                {
                    _bricks.Add(new Brick(20 + col * 36, 25 + row * 12, rowColors[row]));
                }
            }
        }

        public void Run()
        {
            AppleVideo.SetDoubleHiRes(true);
            AppleVideo.Clear(AppleVideo.Black);
            Console.WriteLine("RETRO BREAKOUT ULTRA IN C# ACTIVE!");
            AppleSound.Beep(587, 80);
            AppleSound.Beep(880, 120);

            foreach (var b in _bricks) b.Draw();

            while (_lives > 0 && _bricks.Count > 0)
            {
                int paddleX = AppleSystem.GetMouseX() * 2; // Map paddle 0..510
                AppleVideo.DrawLine(paddleX, 184, paddleX + 48, 184, AppleVideo.White);
                UpdateBall(paddleX);
            }
        }

        private void UpdateBall(int paddleX)
        {
            _ballX += _ballDx;
            _ballY += _ballDy;

            // Wall collisions
            if (_ballX <= 5 || _ballX >= 555) _ballDx = -_ballDx;
            if (_ballY <= 5) _ballDy = -_ballDy;

            // Paddle collision
            if (_ballY >= 180 && _ballX >= paddleX && _ballX <= paddleX + 48)
            {
                _ballDy = -_ballDy;
                AppleSound.Beep(1200, 20);
            }

            // Brick collision
            for (int i = 0; i < _bricks.Count; i++)
            {
                var b = _bricks[i];
                if (b.IsAlive && _ballX >= b.X && _ballX <= b.X + 32 && _ballY >= b.Y && _ballY <= b.Y + 8)
                {
                    b.IsAlive = false;
                    _ballDy = -_ballDy;
                    _score += 100;
                    AppleSound.Beep(880, 15);
                    // Erase brick with black box
                    for (int r = 0; r < 8; r++) AppleVideo.DrawLine(b.X, b.Y + r, b.X + 30, b.Y + r, AppleVideo.Black);
                    break;
                }
            }
        }
    }
}

2. 3D Wireframe Starship Engine (Linear Algebra & Matrix Transformations)

Demonstrates 3D Cartesian coordinates using 8.8 fixed-point integer mathematics, Euler Y-axis rotation matrices, camera perspective projection onto Double Hi-Res raster lines: 

using System;
using Apple2.Ultra;

namespace Starship3DEngine
{
    public struct Vector3D
    {
        public int X, Y, Z;
        public Vector3D(int x, int y, int z) { X = x; Y = y; Z = z; }

        public Vector3D RotateY(int angleDeg)
        {
            double rad = angleDeg * 3.14159 / 180.0;
            int cosA = (int)(Math.Cos(rad) * 256);
            int sinA = (int)(Math.Sin(rad) * 256);
            return new Vector3D((X * cosA - Z * sinA) >> 8, Y, (X * sinA + Z * cosA) >> 8);
        }

        public (int ScreenX, int ScreenY) Project(int focalLength)
        {
            int distance = Z + 300;
            if (distance <= 10) distance = 10;
            return (140 + ((X * focalLength) / distance), 96 - ((Y * focalLength) / distance));
        }
    }
}

3. SmartPort High-Score & Save-State Ledger (ProDOS 32MB Storage)

Demonstrates binary object serialization of a PlayerRecord entity into 512-byte raw ProDOS blocks, invoking AppleStorage.WriteBlock(0x70, blockNum, buffer) to persist game state to the 32MB SmartPort hard drive (/HD): 

using System;
using System.IO;
using Apple2.Ultra;

namespace StorageLedger
{
    public class PlayerRecord
    {
        public string Name { get; set; }
        public int Score { get; set; }
        public byte Level { get; set; }

        public byte[] Serialize()
        {
            byte[] buf = new byte[512];
            for (int i = 0; i < Name.Length && i < 16; i++) buf[i] = (byte)Name[i];
            buf[16] = (byte)(Score & 0xFF);
            buf[17] = (byte)((Score >> 8) & 0xFF);
            buf[18] = Level;
            return buf;
        }
    }

    public class DatabaseEngine
    {
        public static void Main()
        {
            Console.WriteLine("=== PRODOS MLI 32MB SMARTPORT STORAGE ===");
            PlayerRecord hero = new PlayerRecord { Name = "WOZ_CHAMP", Score = 9840, Level = 7 };
            AppleStorage.WriteBlock(AppleStorage.UnitSlot7Drive1, 2, hero.Serialize());
            Console.WriteLine("SAVED TO /HD BLOCK 2 SUCCESSFULLY.");
        }
    }
}

4. Mockingboard & OPL2 FM Multi-Voice Synthesizer

Programs Slot 4 VIA 6522 Port B ($C400) and Port A ($C404) AY-3-8910 tone period and volume registers to generate polyphonic chords: 

using System;
using Apple2.Ultra;

namespace SoundSynthesisEngine
{
    public class PolyphonicSynth
    {
        public static void Main()
        {
            Console.WriteLine("--- SWEET MICRO MOCKINGBOARD DUAL AY-3-8910 ---");
            AppleSound.PlayMockingboard(voice: 0, frequency: 440, volume: 15); // Root A
            AppleSound.PlayMockingboard(voice: 1, frequency: 554, volume: 13); // Third C#
            AppleSound.PlayMockingboard(voice: 2, frequency: 659, volume: 11); // Fifth E
            AppleSound.Beep(880, 100);
        }
    }
}

5. Fractal Math Explorer (16.16 Fixed-Point Arithmetic)

Evaluates the complex quadratic recurrence $Z_{n+1} = Z_n^2 + C$ using 12-bit fractional fixed-point math to render vibrant escape-time fractal color palettes on Double Hi-Res graphics: 

using System;
using Apple2.Ultra;

namespace FractalExplorer
{
    public struct Fixed16
    {
        public int RawValue;
        public Fixed16(int raw) { RawValue = raw; }
        public static Fixed16 operator *(Fixed16 a, Fixed16 b) => new Fixed16((a.RawValue * b.RawValue) >> 12);
    }

    public class MandelbrotEngine
    {
        public static void Main()
        {
            AppleVideo.SetDoubleHiRes(true);
            AppleVideo.Clear(AppleVideo.Black);
            for (int y = 20; y < 180; y += 8) {
                for (int x = 20; x < 260; x += 12) {
                    int col = (x / 20 + y / 16) % 15 + 1;
                    AppleVideo.DrawLine(x, y, x + 8, y, col);
                }
            }
        }
    }
}
