1. The High-Level C# Source: "RetroBreakout"

Object-Oriented Design

Below is the clean C# implementation using classes, encapsulation, inheritance, and the Apple2.Ultra hardware abstraction layer: 

using System;
using Apple2.Ultra;

namespace RetroBreakout
{
    // -------------------------------------------------------------
    // Vector Entity Base Class (Encapsulation & Inheritance)
    // -------------------------------------------------------------
    public abstract class GameObject
    {
        public byte X { get; set; }
        public byte Y { get; set; }

        public abstract void Draw();
    }

    // -------------------------------------------------------------
    // Player Paddle Entity
    // -------------------------------------------------------------
    public class Paddle : GameObject
    {
        public byte Width { get; } = 24;

        public void MoveLeft()  { if (X > 5)  X -= 4; }
        public void MoveRight() { if (X < 110) X += 4; }

        public override void Draw()
        {
            AppleVideo.DrawHLine(X, Y, Width, AppleVideo.ColorWhite);
        }
    }

    // -------------------------------------------------------------
    // Ball Physics Entity
    // -------------------------------------------------------------
    public class Ball : GameObject
    {
        public sbyte VelocityX { get; set; } = 1;
        public sbyte VelocityY { get; set; } = -1;

        public void Update()
        {
            X = (byte)(X + VelocityX);
            Y = (byte)(Y + VelocityY);

            // Wall Collisions
            if (X <= 2 || X >= 136) { VelocityX = (sbyte)-VelocityX; AppleSound.Beep(880, 20); }
            if (Y <= 2)             { VelocityY = (sbyte)-VelocityY; AppleSound.Beep(880, 20); }
        }

        public override void Draw()
        {
            AppleVideo.DrawPixel(X, Y, AppleVideo.ColorYellow);
        }
    }

    // -------------------------------------------------------------
    // Brick Matrix (Array Data Structure)
    // -------------------------------------------------------------
    public class BrickGrid
    {
        public byte[] Bricks = new byte[32]; // 4 rows of 8 bricks

        public void Reset()
        {
            for (byte i = 0; i < 32; i++) Bricks[i] = 1; // 1 = Active
        }

        public bool CheckCollision(byte bx, byte by)
        {
            if (by < 10 || by > 50) return false;
            byte row = (byte)((by - 10) / 10);
            byte col = (byte)(bx / 17);
            byte idx = (byte)(row * 8 + col);

            if (idx < 32 && Bricks[idx] == 1)
            {
                Bricks[idx] = 0; // Destroy brick
                AppleSound.PlayMockingboard(voice: 1, freq: 440, vol: 15);
                return true;
            }
            return false;
        }
    }

    // -------------------------------------------------------------
    // Main Game Controller
    // -------------------------------------------------------------
    public class BreakoutGame
    {
        private Paddle player = new Paddle { X = 60, Y = 180 };
        private Ball ball = new Ball { X = 70, Y = 100 };
        private BrickGrid grid = new BrickGrid();
        private int score = 0;

        public void Run()
        {
            // Initialize 560x192 Double Hi-Res Graphics
            AppleVideo.SetDoubleHiRes(true);
            AppleVideo.ClearScreen(AppleVideo.ColorBlack);
            grid.Reset();

            while (true)
            {
                // 1. Process Input
                char key = AppleInput.GetKey();
                if (key == 'A' || key == 'a') player.MoveLeft();
                if (key == 'D' || key == 'd') player.MoveRight();

                // 2. Update Physics
                ball.Update();

                // 3. Collision with Paddle
                if (ball.Y >= player.Y - 2 && ball.X >= player.X && ball.X <= player.X + player.Width)
                {
                    ball.VelocityY = -1;
                    AppleSound.Beep(523, 30);
                }

                // 4. Collision with Bricks
                if (grid.CheckCollision(ball.X, ball.Y))
                {
                    ball.VelocityY = 1;
                    score += 10;
                }

                // 5. Render Scene
                player.Draw();
                ball.Draw();
            }
        }
    }
}

2. How C# Objects & Fields Live in Apple II Memory

Memory Architecture

In modern systems, an object has a 64-bit reference pointer, object headers, and GC sync blocks. On the 65C02 CPU, the AOT compiler eliminates overhead by compiling objects into flat struct byte offsets with Zero Page fast references:

Zero Page Fast-Register Mapping ($80–$FF)

  • $80:$81: this pointer (16-bit address of current object).
  • $82:$83: player instance memory address ($6000).
  • $84:$85: ball instance memory address ($6010).
  • $86:$87: grid array memory address ($6020).
  • $88:$89: Evaluation operand scratchpad.

65C02 Assembly Output: 
; --- Update Ball X & Y ---
BALL_UPD:
       LDY #$00       ; Offset 0 (X)
       LDA ($84),Y    ; Load Ball.X
       LDY #$02       ; Offset 2 (VelocityX)
       CLC
       ADC ($84),Y    ; X + VelocityX
       LDY #$00
       STA ($84),Y    ; Store back to Ball.X

       LDY #$01       ; Offset 1 (Y)
       LDA ($84),Y    ; Load Ball.Y
       LDY #$03       ; Offset 3 (VelocityY)
       CLC
       ADC ($84),Y    ; Y + VelocityY
       LDY #$01
       STA ($84),Y    ; Store back to Ball.Y

; --- Check X Wall Bounce ---
       LDY #$00
       LDA ($84),Y    ; Load Ball.X
       CMP #$02       ; Left wall collision?
       BMI BOUNCE_X
       CMP #$88       ; Right wall (136)?
       BCC NO_BOUNCE

BOUNCE_X:
       LDY #$02
       LDA ($84),Y    ; Load VelocityX
       EOR #$FF       ; Two's complement negation
       INC A
       STA ($84),Y    ; VelocityX = -VelocityX

       ; Inline Speaker Pulse ($C030)
       JSR SOUND_BEEP_880
NO_BOUNCE:
       RTS
C# Routine: AppleVideo.DrawPixel(X, Y, Color) Hardware Video Blitter @ $2300

The Apple II video RAM is notoriously non-linear. The compiler generates an efficient jump table using the scanline formula: 

; Input: A = X (0..139), X = Y (0..191), Y = Color (0..15)
DRAW_PIXEL:
       PHX            ; Save Y scanline
       PHA            ; Save X coordinate
       
       ; 1. Resolve 16-bit scanline base address from Table
       TXA
       ASL A          ; Multiply Y by 2 for 16-bit table index
       TAX
       LDA HGR_L,X    ; Low address byte ($00, $80, $00...)
       STA $88
       LDA HGR_H,X    ; High address byte ($20, $24, $28...)
       STA $89
       
       ; 2. Determine Main vs Auxiliary Video Bank ($C00D / $C05F)
       PLA            ; Restore X
       LSR A          ; Divide X by 2 (Even X = Aux Bank, Odd X = Main Bank)
       TAY            ; Y register holds screen column offset (0..39)
       BCC USE_AUX

USE_MAIN:
       STA $C004      ; RAMWRTOFF -> Main RAM
       BRA PLOT_BYTE
USE_AUX:
       STA $C005      ; RAMWRTON -> Aux RAM

PLOT_BYTE:
       LDA ($88),Y    ; Load existing pixel byte
       ORA #$7F       ; Apply color mask
       STA ($88),Y    ; Write to Video RAM ($2000-$3FFF)
       STA $C004      ; Reset back to Main RAM write
       PLX
       RTS
C# Routine: AppleSound.PlayMockingboard(voice, freq, vol) Slot 4 6522 VIA Driver @ $2400 
; Programs AY-3-8910 Sound Chip in Slot 4 ($C0C0-$C0CF)
MOCKING_SND:
       LDA #$00       ; Register 0: Tone Period Channel A Fine Tune
       STA $C0C0      ; VIA Port A (Register Select)
       LDA #$07       ; Latch address strobe
       STA $C0C4      ; Control Register
       LDA #$40       ; Frequency Fine Tune Byte
       STA $C0C0      ; Write data byte
       LDA #$06       ; Write strobe
       STA $C0C4
       
       LDA #$08       ; Register 8: Channel A Volume
       STA $C0C0
       LDA #$0F       ; Full Volume (15)
       STA $C0C0
       RTS

4. High Score Persistence to Slinky 1MB RAM & 32MB Hard Disk

Persistent Storage

When the game finishes, high scores are archived into the 1MB Slinky RAM bank or saved as a ProDOS block directly to the 32MB SmartPort hard disk volume: 

; Archive High Score into Slinky RAM Bank $020000 (128KB offset)
SAVE_HIGHSCORE:
       LDA #$00 : STA $C073  ; Slinky Address Low = $00
       LDA #$00 : STA $C074  ; Slinky Address Mid = $00
       LDA #$02 : STA $C075  ; Slinky Address High = $02
       
       ; Stream High Score bytes (Address auto-increments with each write)
       LDA SCORE_L : STA $C071
       LDA SCORE_H : STA $C071
       RTS

5. Modern ProDOS Storage Subsystem & MLI Block Driver Architecture

Object-Oriented Storage Access in C# (.NET Core) and Java 17 for Floppy Diskettes & 32MB SmartPort Hard Disk (/HD)

PRODOS MLI $C700

The Apple //c Ultra bridges modern managed file streams (System.IO.File, java.io.File) and raw block device access (AppleStorage.ReadBlock, AppleStorage.WriteBlock) directly to the vintage ProDOS Machine Language Interface (MLI) Block Device Driver at $C700 (Slot 7 32MB SmartPort Hard Disk) and $C600 (Slot 6 5.25" Floppy Disk Controller).
